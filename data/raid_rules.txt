# Raid rules (src/raid_rules.h, config RaidRulesFile): per raid map the adds to kill first, the casts to interrupt first, the dispel kind of an aura, the ground
# areas to stay out of, the debuff that swaps the tanks, and the no-area-damage fights. Format and the meaning of each row: raid_rules.h and README.md.
#   boss|map|entry|name
#   add|map|entry|rank|name                      rank 0 first
#   interrupt|map|spell_id|name
#   dispel|map|aura_id|magic/curse/disease/poison|name
#   avoid|map|spell/object|id|radius|name
#   swap|map|boss_entry|aura_id|stacks|name
#   noaoe|map|boss_entry|name                    boss_entry 0 = every fight on the map
#   frontal|map|boss_entry|spell_id|name
#   tank_spot|map|boss_entry|x|y|z|yards|name    the boss's victim (a tank) holds it from here, within yards (1..15); RaidBossPositions
#   flank|map|boss_entry|front_deg|rear_deg|rear_yards|name   the rest stand out of the front cone and the rear cone (within rear_yards); RaidBossPositions
#   spread|map|boss_entry|yards|who|name         who: all/ranged/healers/melee; boss_entry 0 = any engaged hostile of the map
#   avoid_enemy|map|entry|yards|who|name         who also tanks (only with RaidAvoidEnemyTank); melee rows act only with RaidAvoidEnemyMelee
#   avoid_enemy_aura|map|entry|yards|who|aura_id|name   as avoid_enemy, only while the hostile carries the aura (seen in a member's target_auras; unknown keeps the circle)
#   purge|map|aura_spell_id|name
#   mindcontrol|map|aura_spell_id|name
# A spell id marked (check) was written from memory and is not yet confirmed against the spell data (none left: see doc/raid-rules.md).
